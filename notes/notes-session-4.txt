type prop_type = {
  // if you want to pass single object then use below one
  // product: {
  //     product_name: string,
  //     brand_name: string,
  //     price: number
  // }

  // let say you want to accept array of object

  product_info: {
    product_name: string;
    brand_name: string;
    price: number;
  }[];
};

function ProductDetails(props: prop_type) {
  return (
    <>
      <h1>Product Details</h1>
      {/* Below one only works when we have single object */}
      {/* <h3>
                Product name: {props.product.product_name}<br></br>
                Brand name: {props.product.brand_name}<br></br>
                Price: {props.product.price}
            </h3> */}

      {/* In case of array of object we need to iterate it. we will use map function, which takse callback.
            As you can see below code(the code written in tbody section basically) is written in {} becase we are writing JS code in between the html code.
            */}
      <table>
        <thead>
          <tr>
            <th>Product Name</th>
            <th>Brand Name</th>
            <th>Price</th>
          </tr>
        </thead>
        <tbody>
          {props.product_info.map((product) => {
            return (
              <tr>
                <td>{product.product_name}</td>
                <td>{product.brand_name}</td>
                <td>{product.price}</td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </>
  );
}

export default ProductDetails;

let say in child you want to do like props.age = 36 ---- this will give you an runtime error.. or try doing any other property from props you will get the error regarding read only property.. so basially you can not assign/modify props as they are read only. they are immutable.

so we want something which can - 

1. it must hold your "state" or "information" or you can say the data relevant to component
2. must be modifiable

so basically... If you go with the props then props is not allowing you to modify and the most important props is something which is coming from the parent. Here we are not looking for something which is coming from the parent. Here we are looking for something which is maintained by the component itself. 

Idea of STATE:

Suppose I'm having. Component like employee details and it is all the employee details. I want something which will help us to maintain that data of that particular component and even we must be in a position to modify it whenever required.

OK, so there is something coming to the picture that is called "states".

OK, so we have learned the props and other is state.

So what is state? State is something which is "responsible to hold the data of relevant specific component" and we can modify it.

so state is mutable. so state is created and maintained in same component.
-----

Previously, state is applicable to only class component. Previously OK state is only applicable to the class component. But now we come up with the hooks. And "hooks" make it possible to apply state in Functional component.

in the old version, we never used to work with a state in a functional component that is most biggest drawback of your functional component in old versions.

That functional component will not able to maintain the state of the data, but now we have a hooks and hooks help us to maintain the state in a functional component and which hook we are having "useState()" the hook which hook we are having useState() hooks. Hooks are nothing but the functions. It is a replacement of the life cycle methods.

import React, {useState} from 'react'

function ProductDetails() {

// let say we wanted to save the quantity of a product a user wants to buy
// we could use useState to manage that state

const [quantity, setQuantity] = useState(10);

function increment_quantity() {
    // see here we can not simly do quantity = quantity + 1
    // we have to use the setQuantity function to update the state becuase if we do we will get an error like you can not assign to a constant variable
  setQuantity(quantity + 1);
}

  return (
    <div>
      Product Quantity: {quantity}
      <br></br>
      {/* <button onClick={increment_quantity}>
        Increase Quantity
      </button> */}

      {/* lets try arrow function method here as this just one liner, no need of seperate function */}
      <button onClick={() => setQuantity(quantity + 1)}>
        Increase Quantity
      </button>
    </div>
  )
}

export default ProductDetails
