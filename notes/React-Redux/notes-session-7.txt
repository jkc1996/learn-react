    }

-----------------------------
And for that we can use, async/await whicwill give use the actual object.

// why Promise<> as return type ?
    //because axios.post is promise based function it will return a promise which will resolve to the actual employee object once the request is completed.
    
    static insertEmployee = async (employee: employee_type): Promise<employee_type> => {
        // Make a POST request to insert a new employee. but we want acual object here not a promise so we use async/await
       let response = await axios.post("http://localhost:3000/employees", employee)
        console.log("Employee inserted:", response.data);
        return response.data; // return the actual inserted employee object
    }

we want the list to be present once the compopnent render:

so when we want to load someting on page render we have a hook called - useEffect(). which takses one callback functon s parammeter 

e.g  useEffect(function x(){
	console.log("I will be called on page load");
})

(or pass and arrow function insted on normal function as callback)

useEffect(()=> {
	console.log("I will be called on page load");
})


**variation with useEffect():

case - 1: simple one:

useEffect(()=> {
	console.log("I will be called on page load");
})

in above case callback that is consoling will happen on evry click of the link.

case - 2: it is like  - useEffect(callback, [dependencies_list])

useEffect(callback, []) - if we do this then the callback function here will execute only once..

but if we dont pass this [] as dependencies list, it will call the callback multiple time.. like let say you are calling this component on some click of the link. so everytime ypu clik on the link (being on the same link), it will execute the callback. but if we pass the [] then it will call the callback only once..

case-3: 

for this case, lets create one useState hook

const [count, setCount] = useState(10)

in the rendering part

<button onClick={setCount(count+1)}> in crement</button>
<button onclick={setCount(count-1)}> decrement</button>

and now with this if we write like this(check console):

useEffect(()=>{console.log("I will get called on every click of the bttons as in the dependencies list 'count' is getting modified")}, [count])

so below is the code to make the get employee details and render it in the tabular form:

import React, { useEffect, useState } from 'react'
import { EmployeeService } from '../../services/EmployeeService';
import { employee_type } from '../../models/employee';

function EmployeeDetails() {

// now lets hold the response using state variable

const [employees, setEmployees] = useState([] as employee_type[]);

  const getAllEmployees = async () => {
    let response = await EmployeeService.fetchEmployees();
    console.log(response);
    setEmployees(response); // setting the response to state variable 'employees' which we will print in the tabular format
}
// below one was just to understand useEffect with dependency_list
const [count, setCount] = useState(0);

// below hook will run only once when the component is mounted sine the dependency_list is empty []
// useEffect(() => {
//     getAllEmployees();
// }, []);

useEffect(() => {
  getAllEmployees();
}, [count]); // here the dependency_list has 'count' state variable, so whenever count changes this useEffect will run

  return (
    <div>
      <h2>Employee Details</h2>
      <p>This is the Employee Details component. check the console for employee Details.</p>

      {/* To understand the dependency_list in the use effect let's create one useState */}
      <h3>{count}</h3>
      {/* so on below button clicks useEffect will get trigged and it will call the getAllEmployees as the count variable is mentiond in the dependecy list of useEffect Hook */}
      <button onClick={() => setCount(count + 1)}>Increment Count</button>
      <button onClick={() => setCount(count - 1)}>Decrement Count</button>

      <h3>Employee List</h3>
      <table border={1}>
        <thead>
          <tr>
            <th>Employee ID</th>
            <th>Employee Name</th>
            <th>Employee Position</th>
            <th>Employee Gender</th>
          </tr>
        </thead>
        <tbody>
          {
            employees.map((employee) => (
              <tr key={employee.id}>
                <td>{employee.id}</td>
                <td>{employee.name}</td>
                <td>{employee.position}</td>
                <td>{employee.gender}</td>
              </tr>
            ))
          }
        </tbody>
      </table>
    </div>
  )
}

export default EmployeeDetails
